নোট:

আপনার জন্য **NilLang**-এ একটি সম্পূর্ণ **নোট অ্যাপ্লিকেশন** তৈরি করে দিচ্ছি। এটি Alap (আলাপ) ফ্রেমওয়ার্কের সবচেয়ে গুরুত্বপূর্ণ ফিচারগুলো ব্যবহার করবে—`@State`, `@Computed`, `async/await`, ফাইল আই/ও, `LazyList`, এবং স্প্লিট-ভিউ লেআউট।

এই অ্যাপটিতে আপনি **নোট তৈরি, এডিট, সেভ এবং ডিলিট** করতে পারবেন। ডেটা লোকাল ফাইল সিস্টেমে `notes.json` ফাইলে সংরক্ষণ করা হবে।

### 📝 সম্পূর্ণ নোট অ্যাপের কোড (NilLang)

```nil
// File: main.nil
// Alap Note App - Full CRUD Application

import { App, Column, Row, Text, Input, Button, LazyList, Icon } from "nil/ui"
import { fs } from "nil/fs"
import { json } from "nil/json"
import { time } from "nil/time"
import { log } from "nil/core"

// ----- ডেটা মডেল -----
struct Note {
    id: i64
    title: string
    content: string
    updatedAt: i64
}

// ----- অ্যাপ্লিকেশন -----
app NotesApp {
    // ----- স্টেট -----
    @State notes: Note[] = []
    @State selectedId: i64? = null
    @State editTitle: string = ""
    @State editContent: string = ""
    @State isNew: bool = false

    // ----- কম্পিউটেড প্রপার্টি -----
    @Computed selectedNote: Note? {
        return notes.find(n => n.id == selectedId)
    }

    @Computed sortedNotes: Note[] {
        return notes.sort((a, b) => b.updatedAt - a.updatedAt)
    }

    // ----- লাইফসাইকেল -----
    onAppear() {
        spawn loadNotes()  // UI থ্রেড ব্লক না করে ব্যাকগ্রাউন্ডে লোড
    }

    // ----- ডেটা পার্সিস্টেন্স (Async) -----
    async function loadNotes() {
        log.info("Loading notes from disk...")
        let data = await fs.readFile("notes.json")
        if data != null {
            let parsed = json.parse<Note[]>(data.toString())
            if parsed != null {
                notes = parsed
                log.info("Loaded " + notes.length + " notes")
            }
        } else {
            // প্রথমবার ব্যবহারের জন্য ডিফল্ট নোট
            notes = [
                Note {
                    id: time.now(),
                    title: "Welcome to Onuron Notes!",
                    content: "Write your thoughts here.",
                    updatedAt: time.now()
                }
            ]
            await saveNotes()
        }
    }

    async function saveNotes() {
        let data = json.stringify(notes, indent: 2)
        await fs.writeFile("notes.json", data)
        log.info("Notes saved.")
    }

    // ----- অ্যাকশন (CRUD) -----
    func createNote() {
        let note = Note {
            id: time.now(),
            title: "Untitled",
            content: "",
            updatedAt: time.now()
        }
        notes.push(note)
        selectedId = note.id
        editTitle = note.title
        editContent = note.content
        isNew = true
        spawn saveNotes()
    }

    func selectNote(id: i64) {
        selectedId = id
        let note = notes.find(n => n.id == id)
        if note != null {
            editTitle = note.title
            editContent = note.content
            isNew = false
        }
    }

    func saveCurrentNote() {
        if selectedId == null { return }

        // যদি টাইটেল ও কন্টেন্ট খালি থাকে, তাহলে নোট ডিলিট করে দিন
        if editTitle.trim() == "" && editContent.trim() == "" {
            deleteNote(selectedId!)
            return
        }

        let index = notes.findIndex(n => n.id == selectedId)
        if index >= 0 {
            notes[index].title = editTitle
            notes[index].content = editContent
            notes[index].updatedAt = time.now()
        }
        isNew = false
        spawn saveNotes()
    }

    func deleteNote(id: i64) {
        notes = notes.filter(n => n.id != id)
        if selectedId == id {
            selectedId = null
            editTitle = ""
            editContent = ""
        }
        spawn saveNotes()
    }

    // ----- ইউজার ইন্টারফেস (UI) -----
    build() {
        Row {
            // ----- সাইডবার: নোটের তালিকা -----
            Column {
                // হেডার
                Row {
                    Text("📒 My Notes")
                        .font(size: 20, weight: .bold)
                    
                    Button("＋ New") {
                        onClick => createNote()
                    }
                    .variant(.primary)
                    .size(width: 80, height: 36)
                }
                .padding(8)
                .spacing(8)
                .width("100%")

                // লেজি লিস্ট (ভার্চুয়ালাইজড স্ক্রল)
                LazyList {
                    items: sortedNotes
                    render: (note) => {
                        Row {
                            Column {
                                Text(note.title)
                                    .font(size: 16, weight: .bold)
                                    .foreground(selectedId == note.id ? .primary : .text)
                                
                                Text(note.content)
                                    .font(size: 12)
                                    .foreground(.secondary)
                                    .maxLines(1)
                            }
                            .grow()
                            .onClick => selectNote(note.id)
                            
                            Button("✕") {
                                onClick => deleteNote(note.id)
                            }
                            .variant(.danger)
                            .size(width: 30, height: 30)
                            .disabled(selectedId == note.id && isNew) // নতুন নোট ডিলিট না
                        }
                        .padding(10)
                        .background(selectedId == note.id ? .primary.opacity(0.12) : .transparent)
                        .cornerRadius(8)
                        .margin(bottom: 4)
                    }
                }
                .grow()
            }
            .width(280)
            .padding(8)
            .background(.surface)
            .border(right: 1, .divider)

            // ----- এডিটর প্যানেল -----
            Column {
                if selectedId != null {
                    // টাইটেল ইনপুট
                    Input(editTitle)
                        .placeholder("Note Title")
                        .font(size: 24, weight: .bold)
                        .padding(bottom: 12)
                        .onChange => editTitle = it

                    // কন্টেন্ট ইনপুট (মাল্টিলাইন)
                    Input(editContent)
                        .placeholder("Start writing your note...")
                        .multiline(true)
                        .grow()
                        .onChange => editContent = it

                    // অ্যাকশন বাটন
                    Row {
                        Button("💾 Save") {
                            onClick => saveCurrentNote()
                        }
                        .variant(.primary)
                        .grow()

                        Button("🗑 Delete") {
                            onClick => deleteNote(selectedId!)
                        }
                        .variant(.danger)
                        .disabled(isNew) // নতুন নোট ডিলিট না
                    }
                    .spacing(8)
                    .padding(top: 12)
                    .width("100%")

                } else {
                    // কোনো নোট সিলেক্ট না থাকলে প্লেসহোল্ডার
                    Column {
                        Icon("📄")
                            .size(64)
                            .foreground(.secondary)
                        
                        Text("Select a note from the sidebar")
                            .font(size: 18)
                            .foreground(.secondary)
                            .padding(top: 16)
                        
                        Text("or create a new one")
                            .font(size: 14)
                            .foreground(.secondary)
                    }
                    .align(.center)
                    .fill()
                }
            }
            .padding(24)
            .grow()
        }
        .fill()
        .background(.background)
    }
}
```

### 🚀 কীভাবে রান করবেন

1. **প্রজেক্ট তৈরি করুন:**
```bash
nil create notes-app
cd notes-app
```

2. **উপরের কোডটি `src/main.nil` ফাইলে পেস্ট করুন।**

3. **Linux এ রান করুন (ডেভেলপমেন্ট):**
```bash
nil run --target linux
```

4. **Onuron এর জন্য বিল্ড করুন:**
```bash
nil build onuron
```

5. **Android এর জন্য বিল্ড করুন:**
```bash
nil build android
```

// File: main.nil
// NilZar Browser - Cross-platform Web Browser for Onuron, Android, iOS & Linux

import { App, Column, Row, Text, Input, Button, Icon, Progress, Stack } from "nil/ui"
import { WebView, WebViewRef, WebViewEvent } from "nil/web"   // Platform WebView Adapter
import { log, time } from "nil/core"
import { fs, path } from "nil/fs"
import { json } from "nil/json"

// ----- ডেটা মডেল (History & Bookmarks) -----
struct HistoryEntry {
    url: string
    title: string
    timestamp: i64
}

struct Bookmark {
    id: string
    name: string
    url: string
    icon: string?
}

// ----- ব্রাউজার অ্যাপ -----
app NilZarBrowser {
    // ----- স্টেট (UI) -----
    @State url: string = "https://onuron.dev"
    @State title: string = "NilZar Browser"
    @State loading: bool = false
    @State progress: f32 = 0.0
    @State canGoBack: bool = false
    @State canGoForward: bool = false
    
    @State showHistory: bool = false
    @State showBookmarks: bool = false
    @State history: HistoryEntry[] = []
    @State bookmarks: Bookmark[] = []
    
    @State searchQuery: string = ""
    @State suggestions: string[] = []

    // ----- ওয়েবভিউ রেফারেন্স -----
    private webViewRef: WebViewRef? = null
    private currentUrl: string = ""

    // ----- ডিফল্ট বুকমার্ক -----
    private defaultBookmarks: Bookmark[] = [
        { id: "1", name: "Onuron", url: "https://github.com/joysriramsarkar/onuron", icon: "🏠" },
        { id: "2", name: "Alap Framework", url: "https://github.com/joysriramsarkar/alap", icon: "⚡" },
        { id: "3", name: "Go Language", url: "https://go.dev", icon: "🐹" },
        { id: "4", name: "GitHub", url: "https://github.com", icon: "🐙" }
    ]

    // ----- লাইফসাইকেল -----
    onAppear() {
        spawn loadHistory()
        spawn loadBookmarks()
    }

    onDispose() {
        spawn saveHistory()
        spawn saveBookmarks()
    }

    // ----- পার্সিস্টেন্স (Async) -----
    async function loadHistory() {
        let data = await fs.readFile("browser_history.json")
        if data != null {
            let parsed = json.parse<HistoryEntry[]>(data.toString())
            if parsed != null { history = parsed }
        }
    }

    async function saveHistory() {
        let data = json.stringify(history, indent: 2)
        await fs.writeFile("browser_history.json", data)
    }

    async function loadBookmarks() {
        let data = await fs.readFile("browser_bookmarks.json")
        if data != null {
            let parsed = json.parse<Bookmark[]>(data.toString())
            if parsed != null { 
                bookmarks = parsed 
            } else {
                bookmarks = defaultBookmarks
            }
        } else {
            bookmarks = defaultBookmarks
            await saveBookmarks()
        }
    }

    async function saveBookmarks() {
        let data = json.stringify(bookmarks, indent: 2)
        await fs.writeFile("browser_bookmarks.json", data)
    }

    // ----- নেভিগেশন কমান্ড -----
    func navigate(urlStr: string) {
        if urlStr == "" { return }
        
        // যদি সার্চ কোয়েরি হয় (ডট বা প্রোটোকল না থাকলে)
        if !urlStr.contains(".") && !urlStr.startsWith("http://") && !urlStr.startsWith("https://") {
            urlStr = "https://duckduckgo.com/?q=" + urlStr.replace(" ", "+")
        } else if !urlStr.startsWith("http://") && !urlStr.startsWith("https://") {
            urlStr = "https://" + urlStr
        }

        currentUrl = urlStr
        url = urlStr
        if webViewRef != null {
            webViewRef!.load(urlStr)
        }
        
        // ব্রাউজিং হিস্টোরি আপডেট
        let entry = HistoryEntry {
            url: urlStr,
            title: title,
            timestamp: time.now()
        }
        history.push(entry)
        if history.length > 1000 {  // সর্বোচ্চ ১০০০ এন্ট্রি
            history = history.slice(history.length - 1000)
        }
        spawn saveHistory()
    }

    func goBack() {
        if webViewRef != null && canGoBack {
            webViewRef!.goBack()
        }
    }

    func goForward() {
        if webViewRef != null && canGoForward {
            webViewRef!.goForward()
        }
    }

    func reload() {
        if webViewRef != null {
            webViewRef!.reload()
        }
    }

    func stopLoading() {
        if webViewRef != null {
            webViewRef!.stop()
        }
    }

    // ----- বুকমার্ক অ্যাকশন -----
    func addBookmark() {
        if currentUrl == "" { return }
        let name = title
        let exists = bookmarks.find(b => b.url == currentUrl)
        if exists == null {
            let bookmark = Bookmark {
                id: time.now().toString(),
                name: name,
                url: currentUrl,
                icon: "🔖"
            }
            bookmarks.push(bookmark)
            spawn saveBookmarks()
        }
        showBookmarks = true
    }

    func removeBookmark(id: string) {
        bookmarks = bookmarks.filter(b => b.id != id)
        spawn saveBookmarks()
    }

    func loadBookmark(url: string) {
        navigate(url)
        showBookmarks = false
    }

    // ----- WebView ইভেন্ট হ্যান্ডলার -----
    func onPageStarted(event: WebViewEvent) {
        loading = true
        progress = 0.0
        log.info("Loading: " + event.url)
        showHistory = false
        showBookmarks = false
    }

    func onPageProgress(event: WebViewEvent) {
        progress = event.progress / 100.0
    }

    func onPageFinished(event: WebViewEvent) {
        loading = false
        progress = 1.0
        currentUrl = event.url
        url = event.url
        title = event.title ?? "NilZar"
        log.info("Loaded: " + event.url)
    }

    func onTitleChange(event: WebViewEvent) {
        if event.title != null && event.title != "" {
            title = event.title!
        }
    }

    func onNavigationChange(event: WebViewEvent) {
        canGoBack = event.canGoBack
        canGoForward = event.canGoForward
    }

    func onUrlChange(event: WebViewEvent) {
        url = event.url
        currentUrl = event.url
    }

    // ----- অটো-সাজেশন (ডামি) -----
    func updateSuggestions(query: string) {
        searchQuery = query
        if query == "" {
            suggestions = []
            return
        }
        // ডামি সাজেশন (রিয়েল ব্রাউজারে সার্চ ইঞ্জিন API কল করবে)
        suggestions = [
            query + " news",
            query + " github",
            query + " docs",
            "https://" + query + ".com"
        ]
    }

    // ----- UI বিল্ডার -----
    build() {
        Column {
            // ---- টাইটেল বার ----
            Row {
                Text("🌐 " + title)
                    .font(size: 14, weight: .medium)
                    .foreground(.secondary)
                    .grow()
                    .padding(left: 8)
                
                Row {
                    Button("📂") { 
                        onClick => showHistory = !showHistory 
                    }
                    .variant(showHistory ? .primary : .ghost)
                    .size(36, 36)

                    Button("⭐") { 
                        onClick => addBookmark() 
                    }
                    .variant(.ghost)
                    .size(36, 36)
                    
                    Button("📖") { 
                        onClick => showBookmarks = !showBookmarks 
                    }
                    .variant(showBookmarks ? .primary : .ghost)
                    .size(36, 36)
                }
                .spacing(2)
            }
            .padding(4, 2)
            .background(.surface)
            .border(bottom: 1, .divider)

            // ---- টুলবার ----
            Row {
                // Back
                Button("◀") { 
                    onClick => goBack() 
                }
                .disabled(!canGoBack)
                .size(40, 40)
                .variant(.ghost)

                // Forward
                Button("▶") { 
                    onClick => goForward() 
                }
                .disabled(!canGoForward)
                .size(40, 40)
                .variant(.ghost)

                // Reload / Stop
                if loading {
                    Button("✕") { 
                        onClick => stopLoading() 
                    }
                    .variant(.danger)
                    .size(40, 40)
                } else {
                    Button("⟳") { 
                        onClick => reload() 
                    }
                    .variant(.ghost)
                    .size(40, 40)
                }

                // Address Bar + Search
                Stack {
                    Input(url)
                        .placeholder("Search or type URL...")
                        .grow()
                        .height(40)
                        .cornerRadius(20)
                        .background(.surface.opacity(0.8))
                        .border(1, .divider)
                        .padding(left: 12, right: 12)
                        .onConfirm => navigate(it)
                        .onChange => updateSuggestions(it)

                    // Dropdown Suggestions
                    if suggestions.length > 0 && !loading {
                        Column {
                            for suggestion in suggestions {
                                Text(suggestion)
                                    .padding(8)
                                    .background(.surface)
                                    .hoverBackground(.primary.opacity(0.1))
                                    .onClick => {
                                        navigate(suggestion)
                                        suggestions = []
                                    }
                            }
                        }
                        .background(.surface)
                        .border(1, .divider)
                        .cornerRadius(8)
                        .shadow(0, 4, 16, .black.opacity(0.2))
                        .margin(top: 44)
                        .position(.absolute)
                        .width("100%")
                    }
                }
                .grow()
                .margin(left: 4, right: 4)

                // Go Button
                Button("Go") {
                    onClick => navigate(url)
                }
                .variant(.primary)
                .size(60, 40)
                .cornerRadius(20)

                // Bookmark / Menu
                Button("⋮") {
                    onClick => showBookmarks = !showBookmarks
                }
                .variant(.ghost)
                .size(40, 40)
            }
            .padding(6, 4)
            .spacing(4)
            .background(.background)
            .border(bottom: 1, .divider)

            // ---- প্রগ্রেস বার ----
            if loading {
                Progress(progress, total: 1.0)
                    .variant(.linear)
                    .color(.accent)
                    .height(3)
                    .animate(.width, duration: 150)
            } else {
                Progress(0, total: 1.0).height(1)
            }

            // ---- সাইড প্যানেল (হিস্ট্রি/বুকমার্ক) ----
            Row {

```nil
// stores/SettingsStore.nil
// কেন্দ্রীয় স্টেট ম্যানেজমেন্ট (স্টোর)

import { fs } from "nil/fs"
import { json } from "nil/json"
import { log, time } from "nil/core"
import { defaultSettings, AppSettings } from "../models/Settings"

store SettingsStore {
    // ---------- স্টেট ----------
    @State settings: AppSettings = defaultSettings()
    @State searchQuery: string = ""
    @State isLoading: bool = false
    @State isDirty: bool = false
    @State lastSaved: i64 = 0

    // ---------- কম্পিউটেড ----------
    @Computed filteredSections: string[] {
        if searchQuery == "" {
            return allSections
        }
        let query = searchQuery.toLowerCase()
        return allSections.filter(section => {
            return section.toLowerCase().contains(query) ||
                   getSectionItems(section).any(item =>
                       item.label.toLowerCase().contains(query) ||
                       item.description.toLowerCase().contains(query)
                   )
        })
    }

    @Computed isDarkMode: bool {
        if settings.appearance.theme == "dark" {
            return true
        } else if settings.appearance.theme == "light" {
            return false
        } else {
            // System theme - platform API কল করবে
            return Platform.getSystemTheme() == .Dark
        }
    }

    // ---------- লাইফসাইকেল ----------
    onLoad() {
        spawn load()
    }

    // ---------- পার্সিস্টেন্স ----------
    async function load() {
        isLoading = true
        let path = "settings.json"
        let exists = await fs.exists(path)
        if exists {
            let data = await fs.readFile(path)
            if data != null {
                let parsed = json.parse<AppSettings>(data.toString())
                if parsed != null {
                    settings = parsed
                    log.info("Settings loaded from disk")
                }
            }
        }
        isLoading = false
        lastSaved = time.now()
    }

    async function save() {
        let data = json.stringify(settings, indent: 2)
        await fs.writeFile("settings.json", data)
        isDirty = false
        lastSaved = time.now()
        log.info("Settings saved")
    }

    // ---------- সেটার অ্যাকশন ----------
    @Action func setGeneral(general: GeneralSettings) {
        settings.general = general
        isDirty = true
        spawn save()
    }

    @Action func setAppearance(appearance: AppearanceSettings) {
        settings.appearance = appearance
        isDirty = true
        spawn save()
        // থিম পরিবর্তন হলে UI আপডেটের জন্য নোটিফাই
        notifyThemeChanged()
    }

    @Action func setNotifications(notifications: NotificationSettings) {
        settings.notifications = notifications
        isDirty = true
        spawn save()
    }

    @Action func setPrivacy(privacy: PrivacySettings) {
        settings.privacy = privacy
        isDirty = true
        spawn save()
    }

    @Action func setAccount(account: AccountSettings) {
        settings.account = account
        isDirty = true
        spawn save()
    }

    // ---------- সিংগেল ফিল্ড আপডেট (হেল্পার) ----------
    @Action func updateGeneral(field: string, value: any) {
        let mut g = settings.general
        g[field] = value
        settings.general = g
        isDirty = true
        spawn save()
    }

    @Action func updateAppearance(field: string, value: any) {
        let mut a = settings.appearance
        a[field] = value
        settings.appearance = a
        isDirty = true
        spawn save()
    }

    // ---------- রিসেট ----------
    @Action func resetAll() {
        settings = defaultSettings()
        isDirty = true
        spawn save()
    }

    @Action func resetSection(section: string) {
        let defaultSettings = defaultSettings()
        match section {
            "General" => settings.general = defaultSettings.general
            "Appearance" => settings.appearance = defaultSettings.appearance
            "Notifications" => settings.notifications = defaultSettings.notifications
            "Privacy" => settings.privacy = defaultSettings.privacy
            "Account" => settings.account = defaultSettings.account
            _ => log.warn("Unknown section: " + section)
        }
        isDirty = true
        spawn save()
    }

    // ---------- প্রাইভেট মেথড ----------
    private func notifyThemeChanged() {
        // থিম চেঞ্জের জন্য ইভেন্ট ফায়ার
        EventBus.emit("themeChanged", isDarkMode)
    }

    // ---------- হেল্পার ----------
    func getSectionItems(section: string): SettingItem[] {
        // সেকশন অনুযায়ী আইটেম ফেরত দেয়
        // বিস্তারিত নিচে
        return []
    }
}

// ---------- এভেন্ট বাস (সিম্পল) ----------
object EventBus {
    private let listeners: Map<string, (any) => void> = new Map()

    fun emit(event: string, data: any) {
        let listener = listeners.get(event)
        if listener != null {
            listener(data)
        }
    }

    fun on(event: string, callback: (any) => void) {
        listeners.set(event, callback)
    }
}
```

```nil
    build() {
        Column {
            // ---------- হেডার ----------
            Row {
                Text("⚙️ Settings")
                    .font(size: 28, weight: .bold)
                    .grow()

                // সার্চ টগল
                Button("🔍") {
                    onClick => showSearch = !showSearch
                }
                .variant(showSearch ? .primary : .ghost)
                .size(40, 40)

                // সেভ স্ট্যাটাস
                if store.isDirty {
                    Text("⚠️ Unsaved")
                        .font(size: 12)
                        .foreground(.warning)
                        .padding(right: 8)
                }
            }
            .padding(16)
            .border(bottom: 1, .divider)

            // ---------- সার্চ বার ----------
            if showSearch {
                SearchBar {
                    query: store.searchQuery
                    onChange: (q) => store.searchQuery = q
                    onClear: () => store.searchQuery = ""
                }
                .padding(8)
            }

            // ---------- মূল কন্টেন্ট ----------
            Row {
                // ---------- সাইডবার (সেকশন লিস্ট) ----------
                Column {
                    Text("Categories")
                        .font(size: 14, weight: .bold)
                        .foreground(.secondary)
                        .padding(8)

                    LazyList {
                        items: sections
                        render: (section) => {
                            Button(section) {
                                onClick => selectedSection = section
                            }
                            .variant(selectedSection == section ? .primary : .ghost)
                            .align(.left)
                            .padding(8)
                            .width("100%")
                            .cornerRadius(8)
                        }
                    }
                    .grow()

                    // ফুটার
                    Text("v" + store.settings.about.version)
                        .font(size: 12)
                        .foreground(.secondary)
                        .padding(8)
                }
                .width(200)
                .background(.surface)
                .border(right: 1, .divider)

                // ---------- কন্টেন্ট এরিয়া ----------
                Scroll {
                    Column {
                        // লোডিং স্টেটাস
                        if store.isLoading {
                            Text("Loading settings...")
                                .foreground(.secondary)
                                .padding(40)
                        } else {
                            // সার্চ রেজাল্ট
                            if store.searchQuery != "" {
                                Text("Search results for: \"" + store.searchQuery + "\"")
                                    .font(size: 16, weight: .bold)
                                    .padding(8)
                            }

                            // ক্যাটাগরি রেন্ডার
                            let items = getSectionItems(selectedSection)
                            SettingCategory {
                                title: selectedSection
                                icon: getSectionIcon(selectedSection)
                                items: items
                            }
                            .padding(16)
                        }
                    }
                    .padding(8)
                }
                .grow()
                .background(.background)
            }
            .grow()

            // ---------- ফুটার (অ্যাকশন বাটন) ----------
            Row {
                Text("Last saved: " + time.fromTimestamp(store.lastSaved).toLocaleString())
                    .font(size: 12)
                    .foreground(.secondary)
                    .grow()

                Button("💾 Save Now") {
                    onClick => store.save()
                }
                .variant(.primary)
                .disabled(!store.isDirty)

                Button("🔄 Reset All") {
                    onClick => store.resetAll()
                }
                .variant(.danger)
                .size(120, 40)
            }
            .padding(16)
            .border(top: 1, .divider)
            .background(.surface)
        }
        .fill()
        .background(.background)
        .theme(store.isDarkMode ? .dark : .light)
    }

    // ---------- হেল্পার: সেকশন আইকন ----------
```

```yaml
name: com.onuron.settings
```

## ৮. রান করার নির্দেশনা

```bash
# প্রজেক্ট তৈরি
nil create settings-app
cd settings-app

# উপরের সব ফাইল যথাস্থানে রাখুন

# Linux এ রান (ডেভেলপমেন্ট)
nil run --target linux

# অ্যান্ড্রয়েড বিল্ড
nil build android

# Onuron বিল্ড
nil build onuron
```
